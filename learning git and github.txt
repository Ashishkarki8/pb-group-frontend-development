# Create and switch to a new branch

Branch naming convention:
feature/ → new features
bugfix/ → fixing bugs
hotfix/ → urgent production fixes
chore/ → setup, config, docs







# 1. Start fresh from the day
git checkout main       //To switch to the main branch
git pull origin main    

# 2. Create feature branch
git checkout -b feature/your-feature

# 3. Make changes, commit
git add .
git commit -m "feat: your descriptive message"

# 4. Push branch
git push origin feature/your-feature

# 5. Create PR on GitHub
# 6. Wait for review & merge
# 7. Delete branch (optional)
git branch -d feature/your-feature




# BAD - adds EVERYTHING (might include junk)
git add .  ← Avoid in real work!

# GOOD - add specific files
git add frontend/src/App.jsx
git add frontend/src/components/Login.jsx

# ALSO GOOD - add all changes in a folder
git add frontend/src/  # All changes in src folder

# BEST - check what you're adding first
git status  # See exactly what files changed
git add file1.js file2.js  # Add only what you need

# 1. SAVE YOUR PROGRESS (even if incomplete!)
git add .                          # Stage your changes
git commit -m "wip: login form UI in progress"  
# "wip" = Work In Progress - perfectly acceptable!

# 2. GET THE LATEST CODE
git checkout main                 # Switch to main
git pull origin main              # Get Sarah's merged code
git checkout feature/login-button # Back to your branch

# 3. MERGE MAIN INTO YOUR BRANCH
git merge main                    # Bring Sarah's changes into your branch
# (Fix any conflicts if they appear)

# 4. CONTINUE CODING
# Now you have YOUR code + LATEST main code together!

# 5. WHEN FINISHED
git add .
git commit -m "feat: complete login page with validation"
git push origin feature/login-button
